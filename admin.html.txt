let list = [];

// 🔥 LOAD REALTIME
refDB.on("value", snap => {
  list = snap.val() || [];
  updateCount();
});

// 🔥 UI
function toast(msg) {
  const t = document.getElementById("toast");
  t.innerText = msg;
  t.classList.add("show");
  setTimeout(()=>t.classList.remove("show"), 3000);
}

function updateCount() {
  document.getElementById("count").innerText = "📊 Còn: " + list.length;
}

function copy(text) {
  navigator.clipboard.writeText(text);
  toast("✅ Copy: " + text);
}

// ➕ ADD
document.getElementById("add").onclick = () => {
  let val = document.getElementById("input").value;
  let arr = val.split(/[\s,;|]+/).filter(x => x.includes("@"));

  refDB.transaction(data => {
    data = data || [];
    arr.forEach(e => { if (!data.includes(e)) data.push(e); });
    return data;
  });

  document.getElementById("input").value = "";
};

// 🗑️ DELETE
document.getElementById("deleteListButton").onclick = () => {
  let current = document.getElementById("output").value;
  if (!current) return;

  refDB.transaction(data => data.filter(x => x !== current));
  document.getElementById("output").value = "";
};

// 🔁 NEXT
document.getElementById("next").onclick = () => {
  if (list.length == 0) return;

  let cur = document.getElementById("output").value;
  let next = list[0];

  if (cur) {
    let i = list.indexOf(cur);
    next = list[(i + 1) % list.length];
  }

  document.getElementById("output").value = next;
  copy(next);
};

// 📋 COPY BUTTON
document.getElementById("copyBtn").onclick = () => {
  let val = document.getElementById("output").value;
  if (val) copy(val);
};

// 📋 AUTO COPY
document.getElementById("copyButton").onclick = () => {
  if (list.length == 0) return;

  let val = document.getElementById("output").value || list[0];
  document.getElementById("output").value = val;
  copy(val);
};
